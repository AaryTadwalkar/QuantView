## 2. Relational Append-Only Triggers

```sql
-- PostgreSQL Append-Only Enforcement
CREATE OR REPLACE FUNCTION trg_prevent_ledger_mutation()
RETURNS TRIGGER AS $$
BEGIN
    RAISE EXCEPTION 'ledger_events is append-only. UPDATE and DELETE operations are forbidden.';
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_ledger_events_append_only
BEFORE UPDATE OR DELETE ON ledger_events
FOR EACH ROW
EXECUTE FUNCTION trg_prevent_ledger_mutation();

-- SQLite Relational Mirror
CREATE TRIGGER IF NOT EXISTS trg_prevent_ledger_update
BEFORE UPDATE ON ledger_events
BEGIN
    SELECT RAISE(ABORT, 'ledger_events is append-only. UPDATE is forbidden.');
END;

CREATE TRIGGER IF NOT EXISTS trg_prevent_ledger_delete
BEFORE DELETE ON ledger_events
BEGIN
    SELECT RAISE(ABORT, 'ledger_events is append-only. DELETE is forbidden.');
END;
```
